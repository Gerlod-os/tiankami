### 7.1. slugify.js

```js
export function slugify(title) {
  return title
    .toLowerCase()
    .replace(/[^\wа-яёА-Яё\s-]/g, "")
    .replace(/\s+/g, "-")
    .replace(/-+/g, "-")
    .replace(/^-+|-+$/g, "");
}

export function uniqueSlug(slug, usedSlugs) {
  if (!usedSlugs.has(slug)) return slug;
  let i = 2;
  const base = slug;
  while (usedSlugs.has(`${base}-${i}`)) i++;
  return `${base}-${i}`;
}
```

### 7.2. date.js — parseRuDate (критично!)

```js
export function parseRuDate(dateStr) {
  if (!dateStr || !dateStr.trim()) return null;
  const s = dateStr.trim();
  const ruMatch = s.match(/^(\d{1,2})\.(\d{1,2})\.(\d{4})$/);  // DD.MM.YYYY
  if (ruMatch) {
    const [_, day, month, year] = ruMatch;
    return createSafeDate(parseInt(year), parseInt(month), parseInt(day));
  }
  const isoMatch = s.match(/^(\d{4})-(\d{2})-(\d{2})$/);  // YYYY-MM-DD
  if (isoMatch) {
    const [_, year, month, day] = isoMatch;
    return createSafeDate(parseInt(year), parseInt(month), parseInt(day));
  }
  return null;
}
function createSafeDate(year, month, day) {
  const d = new Date(Date.UTC(year, month - 1, day));
  if (isNaN(d.getTime())) return null;
  if (d.getUTCDate() !== day || d.getUTCMonth() !== month - 1 || d.getUTCFullYear() !== year) return null;
  return d;
}
```

Поиск индексов колонок из строки заголовков:
```js
export function buildColIndex(headerRow, firstDataRow) {
  const idx = {};
  headerRow.forEach((cell, i) => {
    const key = (cell || "").trim();
    if (key) idx[key] = i;
  });

  // Название: сначала явный заголовок, потом эвристика
  let titleIdx = -1;
  let titleWasShifted = false;
  if (idx["Название"] !== undefined) {
    titleIdx = idx["Название"];
    // merged cells: заголовок в B, данные в C → если в колонке заголовка данных нет, +1
    if (firstDataRow && firstDataRow[titleIdx] !== undefined && !firstDataRow[titleIdx].trim()) {
      titleIdx++;
      titleWasShifted = true;
    }
  } else {
    // Эвристика: пустой заголовок + есть данные; пропускаем картинку (URL/=IMAGE)
    for (let i = 0; i < headerRow.length; i++) {
      const headerEmpty = !headerRow[i] || !headerRow[i].trim();
      const dataExists = firstDataRow && firstDataRow[i] && firstDataRow[i].trim() !== "";
      if (headerEmpty && dataExists) {
        const firstData = firstDataRow[i].trim();
        if (/^https?:\/\//i.test(firstData) || /^=IMAGE/i.test(firstData)) continue;
        titleIdx = i;
        break;
      }
    }
  }
  if (titleIdx === -1) titleIdx = 2; // fallback

  const imageIdx = titleWasShifted && titleIdx > 1 ? titleIdx - 2 : titleIdx > 0 ? titleIdx - 1 : -1;

  return {
    title: titleIdx,
    image: imageIdx,
    genre: idx["Жанр"] ?? -1,
    features: idx["Особенности"] ?? -1,
    setting: idx["Сеттинг"] ?? -1,
    complexity: idx["Сложность"] ?? -1,
    hours: idx["Наиграно (часы)"] ?? -1,
    status: idx["Статус"] ?? -1,
    progress: idx["Прогресс (%)"] ?? -1,
    rating: idx["Оценка (?/10)"] ?? -1,
    releaseDate: idx["Дата выхода"] ?? -1,
    playedDate: idx["Когда играл"] ?? -1,
    notes: idx["Примечания"] ?? -1,
    youtube: idx["YouTube прохождение"] ?? -1,
    hasMI: idx["Проводил МИ"] ?? -1,
    miVideo: idx["Выпуск МИ"] ?? -1,
  };
}
```

```js
export function normalizeGames(rows) {
  if (!rows || rows.length < 2) return [];
  const headerRow = rows[0];
  const colIndex = buildColIndex(headerRow, rows[1]);
  const games = [];
  for (let i = 1; i < rows.length; i++) {
    const game = normalizeGameRow(rows[i], colIndex);
    if (game) games.push(game);
  }
  games.sort((a, b) => a.title.localeCompare(b.title, "ru"));
  const usedSlugs = new Set();
  return games.map((game) => {
    const baseSlug = slugify(game.title);
    const slug = uniqueSlug(baseSlug, usedSlugs);
    usedSlugs.add(slug);
    return { ...game, slug };
  });
}
```

**Подборки — ПОСТРОЧНЫЙ парсер (не переписывать!):**
```js
export function normalizeCollections(rows) {
  if (!rows || rows.length < 2) return [];
  const headerRow = rows[0];
  const isNumeric = (s) => /^\d+([.,]\d+)?$/.test((s || "").trim());
  const WINDOW_SIZE = 5;   // ширина окна подборки
  const RANK_RANGE = 4;    // поиск ранга правее игры
  const titleIndices = [];
  headerRow.forEach((cell, idx) => {
    if (cell && cell.trim() !== "") titleIndices.push(idx);
  });
  const collections = [];
  const usedSlugs = new Set();
  titleIndices.forEach((titleIdx) => {
    const name = headerRow[titleIdx].trim();
    if (!name) return;
    const minCol = titleIdx;
    const maxCol = titleIdx + WINDOW_SIZE;
    let description = "";
    const underTitle = ((rows[1] || [])[titleIdx] || "").trim();
    if (underTitle && !isNumeric(underTitle)) description = underTitle;
    const games = [];
    for (let i = 1; i < rows.length; i++) {
      const row = rows[i] || [];
      let gameCol = -1;
      for (let c = minCol; c <= maxCol; c++) {
        const v = (row[c] || "").trim();
        if (!v || isNumeric(v)) continue;
        if (i === 1 && c === titleIdx) continue; // описание под заголовком
        gameCol = c;
        break;
      }
      if (gameCol === -1) continue;
      let rank = "";
      for (let c = gameCol + 1; c <= gameCol + RANK_RANGE; c++) {
        const v = (row[c] || "").trim();
        if (v && isNumeric(v)) { rank = v; break; }
      }
      games.push({ name: (row[gameCol] || "").trim(), rank });
    }
    if (games.length === 0) return;
    if (/золотой список/i.test(name)) {
      games.forEach((g) => (g.rank = ""));  // ЗОЛОТОЙ СПИСОК без рангов
    }
    const baseSlug = slugify(name);
    const slug = uniqueSlug(baseSlug, usedSlugs);
    usedSlugs.add(slug);
    collections.push({ name, slug, description, games });
  });
  return collections;
}
```

```js
export function isUrl(value) {
  return /^https?:\/\//i.test((value || "").trim());
}

export function extractSteamAppId(url) {
  const m = (url || "").match(/store\.steampowered\.com\/app\/(\d+)/i);
  return m ? m[1] : null;
}

function extractUrlFromCell(cell) {
  const s = (cell || "").toString().trim();
  if (!s) return null;
  const m = s.match(/^=HYPERLINK\(\s*"([^"]+)"\s*[;,]/i);
  if (m) return m[1];
  if (s.startsWith("http")) return s;
  return null;
}

// Копия, колонки C:W → C=0, U=18 (Steam), V=19 (YouTube), W=20 (МИ)
export function extractLinksFromCopyRow(row) {
  const title = (row[0] || "").toString().trim();
  if (!title) return null;
  const entry = {};
  const steamUrl = extractUrlFromCell(row[18]);
  if (steamUrl) {
    const appId = extractSteamAppId(steamUrl);
    if (appId) entry.steamAppId = appId;
  }
  const ytUrl = extractUrlFromCell(row[19]);
  if (ytUrl) entry.youtube = ytUrl;
  const miUrl = extractUrlFromCell(row[20]);
  if (miUrl) entry.miVideo = miUrl;
  return entry.steamAppId || entry.youtube || entry.miVideo ? { title, entry } : null;
}

export function getGameMetadata(games) {
  const genres = new Set();
  const years = new Set();
  games.forEach((game) => {
    (game.genre || "").split(",").forEach((g) => {
      const t = g.trim();
      if (t) genres.add(t);

  return { genres: [...genres].sort(), years: [...years].sort() };
```

## 8. ПОТОК ДАННЫХ (SWR) — src/utils/loadData.js

Стратегия stale-while-revalidate:
1. **Мгновенно**: память (`memoryGames`) → localStorage-кэш (6 часов) → локальный `src/data/*.json`.
2. **В фоне**: сверка с Google Таблицей (не чаще 1 раза в 15 минут).
3. Если таблица изменилась → обновить кэш и вызвать `onUpdate(свежие)` → UI без перезагрузки.
4. Ключи кэша привязаны к URL таблицы (`hashUrl`) — смена таблицы сбрасывает кэш сам.
5. Восстановление `steamAppId` + обложки из локального JSON, если Sheets API недоступен.

Кэш: `localStorage`, ключи `tiankami_games_v5_{hash}` (+ `_time`), `tiankami_collections_v5_*`, `tiankami_schedule_v5_*`.

Обогащение игр из локального JSON:
```js
// steamAppId → обложка (header) + steamUrl
const appIdBySlug = {};
localGames.forEach((lg) => {
  if (lg.slug && lg.steamAppId) appIdBySlug[lg.slug] = lg.steamAppId;
});
return games.map((g) => {
  const appId = g.steamAppId || appIdBySlug[g.slug];
  if (!appId) return g;
  return { ...g, steamAppId: appId, image: steamHeaderUrl(appId), steamUrl: `https://store.steampowered.com/app/${appId}/` };
});
```

`clearCache()` — сброс всех localStorage кэшей и памяти (кнопка «Обновить данные»).

## 9. СКРИПТ СИНХРОНИЗАЦИИ — scripts/sync-data.js

Логика `npm run sync`:
1. Прочитать `.env` вручную (в Node `import.meta.env` недоступен).
2. Скачать `GAMES_URL` (TSV) → `normalizeGames()` → `games.json`.
3. Обогатить: скачать Копию (Sheets API, `SHEETS_API_KEY_SERVER`, колонки `C:W`,
   `valueRenderOption=UNFORMATTED_VALUE`) → `extractLinksFromCopyRow` → дополнить
   `steamAppId`, `youtube`, `miVideo` по совпадению названия.
4. Скачать `COLLECTIONS_URL` (CSV) → `normalizeCollections()` → `collections.json`.
5. Скачать `SCHEDULE_URL` (CSV с заголовками) → фильтр `r.date && r.game` →
   конвертировать `YYYY-MM-DD` → `DD.MM.YYYY` → сортировка по дате → `schedule.json`.
6. Таймаут запросов 15 сек, 3 попытки. При ошибке — предупреждение, не падение.

```javascript
function extractLink(sheet, row, col) {
  const range = sheet.getRange(row, col);
  let url = '';
  try { url = range.getRichTextValue().getLinkUrl() || ''; if (url) return url; } catch (e) {}
  try {
    const formula = range.getFormula();
    if (formula && formula.toLowerCase().startsWith('=hyperlink')) {
      const m = formula.match(/=HYPERLINK\(\s*"([^"]+)"/i);
      if (m && m[1]) return m[1];
      const m2 = formula.match(/=HYPERLINK\(\s*'([^']+)'/i);
      if (m2 && m2[1]) return m2[1];
    }
  } catch (e) {}
  try {
    const val = String(range.getValue() || '');
    const m = val.match(/(https?:\/\/[^\s<>"{}[\]|]+)/);
    if (m) return m[1];
  } catch (e) {}
  return '';
}
```

## 11. API РАСПИСАНИЯ — api/schedule.js (Vercel serverless)

Назначение: прокси CORS. Браузер шлёт POST на свой домен, Vercel пересылает в Google Apps Script.

```js
export default async function handler(req, res) {
  if (req.method !== "POST") {
    return res.status(405).json({ success: false, error: "Method not allowed" });
  }
  let rawBody = "";
  for await (const chunk of req) rawBody += chunk;
  let body;
  try { body = JSON.parse(rawBody); } catch {
    return res.status(400).json({ success: false, error: "Invalid JSON" });
  }
  const { date, time, game, streamLink } = body;
  if (!date || !time || !game) {
    return res.status(400).json({ success: false, error: "Не все поля заполнены (date, time, game обязательны)" });
  }
  try {
    const response = await fetch(SCRIPT_URL, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ date, time, game, streamLink: streamLink || "https://twitch.tv/tiankami" }),
    });
    if (!response.ok) {
      return res.status(response.status).json({ success: false, error: `Google Apps Script вернул ${response.status}` });
    }
    return res.status(200).json({ success: true });
  } catch (error) {
    console.error("[Proxy] Ошибка:", error);
    return res.status(500).json({ success: false, error: error.message || "Ошибка сервера" });
  }
}
```

## 13. ЖЁСТКИЕ ПРАВИЛА (границы)

1. **Steam API** — не запрашивать! Только CDN-обложки по `steamAppId`:
   `https://cdn.cloudflare.steamstatic.com/steam/apps/{appId}/header.jpg`.
2. **noembed** — максимум 20 проверок за сессию (авторство YouTube).
3. **Кэш автоповторов**: Twitch-статус 5 мин, аватар 24 ч, YouTube-авторство бессрочно.
4. **Подборки** — парсер ПОСТРОЧНЫЙ (колонки «гуляют»), не переписывать на колоночный.
5. **Даты** — только через `parseRuDate` (DD.MM.YYYY и YYYY-MM-DD).
6. **Статус** — «Жду релиз»/«жду» → «Жду релиза» (`normalizeStatus`).
7. **Slug** — уникальный на каждую игру (генерируется в `normalizeGames`).
8. **Гиперссылки** теряются в TSV — достаются из Копии (Sheets API).
9. **Конфиги**: не удалять `vite.config.js`, `package.json`, `vercel.json`, `.env`.
10. **`import.meta.env` в Node** — читать `.env` вручную.
11. **Vite build** — `manualChunks` функцией, не объектом (Vite 8 = rolldown).
12. **Google Fonts** заблокирован в РФ — fallback на system-ui.
